-- Users table
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    role VARCHAR(10) CHECK (role IN ('tenant', 'landlord')),
    name VARCHAR(255) NOT NULL,
    date_of_birth DATE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    address TEXT,               -- Landlord only
    nin VARCHAR(50),            -- Landlord only
    marital_status VARCHAR(20), -- Tenant only
    passport_photo_url TEXT,    -- Landlord only
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Properties table
CREATE TABLE properties (
    id SERIAL PRIMARY KEY,
    landlord_id INT REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(255),
    description TEXT,
    location VARCHAR(255),
    price NUMERIC(12, 2),
    lease_duration_months INT,
    is_occupied BOOLEAN DEFAULT FALSE,
    ownership_certificate_token VARCHAR(255),
    rent_expiry_date DATE,
    verified BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Property Images table (for multiple images per property)
CREATE TABLE property_images (
    id SERIAL PRIMARY KEY,
    property_id INT REFERENCES properties(id) ON DELETE CASCADE,
    image_url TEXT NOT NULL
);

-- Messages table
CREATE TABLE messages (
    id SERIAL PRIMARY KEY,
    sender_id INT REFERENCES users(id) ON DELETE CASCADE,
    recipient_id INT REFERENCES users(id) ON DELETE CASCADE,
    property_id INT REFERENCES properties(id) ON DELETE CASCADE,
    message TEXT NOT NULL,
    sent_at TIMESTAMP DEFAULT NOW()
);

-- Admin Verifications table (to track verification actions)
CREATE TABLE admin_verifications (
    id SERIAL PRIMARY KEY,
    property_id INT REFERENCES properties(id) ON DELETE CASCADE,
    admin_id INT REFERENCES users(id),
    verified BOOLEAN,
    verified_at TIMESTAMP DEFAULT NOW()
);
